   100→/* ===================================================================== *
   101→ *  SystemView — top-level router for the System module
   102→ * ===================================================================== */
   103→
   104→export function SystemView({ moduleId, childId, grandchildId }: ViewProps) {
   105→  const router = useViewRouter(moduleId, childId, grandchildId);
   106→
   107→  if (router.state === "loading") return null;
   108→  if (router.state === "module-overview")
   109→    return <SystemOverview moduleId={moduleId} />;
   110→  if (router.state === "child-overview")
   111→    return <ChildOverview moduleId={moduleId} childId={router.childId} />;
   112→
   113→  if (router.state === "grandchild") {
   114→    /* ---------------- Network ---------------- */
   115→    if (childId === "network" && grandchildId === "interface")
   116→      return <InterfacePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   117→    if (childId === "network" && grandchildId === "gateway")
   118→      return <GatewayPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   119→    if (childId === "network" && grandchildId === "priorities")
   120→      return <PrioritiesPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   121→    if (childId === "network" && grandchildId === "dns")
   122→      return <DnsPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   123→    if (childId === "network" && grandchildId === "static-route")
   124→      return <StaticRoutePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   125→
   126→    /* ---------------- Firewall ---------------- */
   127→    if (childId === "firewall" && grandchildId === "create")
   128→      return <FirewallCreatePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   129→    if (childId === "firewall" && grandchildId === "manage")
   130→      return <FirewallManagePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   131→    if (childId === "firewall" && grandchildId === "dos-settings")
   132→      return <DosSettingsPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   133→    if (childId === "firewall" && grandchildId === "dos-bypass")
   134→      return <DosBypassPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   135→    if (childId === "firewall" && grandchildId === "free-sites")
   136→      return <FreeSitesPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   137→
   138→    /* ---------------- DHCP ---------------- */
   139→    if (childId === "dhcp" && grandchildId === "manage-dhcp")
   140→      return <ManageDhcpPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   141→    if (childId === "dhcp" && grandchildId === "ip-leasing")
   142→      return <IpLeasingPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   143→
   144→    /* ---------------- PPPoE ---------------- */
   145→    if (childId === "pppoe" && grandchildId === "manage-pppoe")
   146→      return <ManagePppoePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   147→
   148→    /* ---------------- Manage Data ---------------- */
   149→    if (childId === "manage-data" && grandchildId === "backup")
   150→      return <BackupPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   151→    if (childId === "manage-data" && grandchildId === "backup-schedule")

   153→    if (childId === "manage-data" && grandchildId === "restore")
   154→      return <RestorePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;

   163→
   164→    /* ---------------- Client Services ---------------- */
   165→    if (childId === "client-services" && grandchildId === "parameters")
   166→      return <ClientServicesParametersPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   167→    if (childId === "client-services" && grandchildId === "customized-images")
   168→      return <CustomizedImagesPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;

   174→      return <WebserviceConfigPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   175→    if (childId === "client-services" && grandchildId === "password-config")

   195→
   196→    /* ---------------- Captive Portal ---------------- */
   197→    if (childId === "captive-portal" && grandchildId === "create")
   198→      return <CaptiveCreatePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;

   212→      return <ConfigureProfilePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   213→    if (childId === "captive-portal" && grandchildId === "security-config")
   214→      return <CpSecurityConfigPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   215→    if (childId === "captive-portal" && grandchildId === "social-media")
   216→      return <SocialMediaConfigPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;

   218→      return <CpRegistrationPolicyPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   219→
   220→    /* ---------------- NAS Management ---------------- */
   221→    if (childId === "nas" && grandchildId === "nas-ip-config")
   222→      return <NasIpConfigPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   223→    if (childId === "nas" && grandchildId === "radius-config")
   224→      return <RadiusConfigPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   225→    if (childId === "nas" && grandchildId === "nas-configuration")
   226→      return <NasConfigurationPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   227→    if (childId === "nas" && grandchildId === "preferences")
   228→      return <NasPreferencesPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;

   242→      return <DeviceLogsPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   243→
   244→    /* ---------------- System Tools ---------------- */
   245→    if (childId === "system-tools" && grandchildId === "packet-capture")
   246→      return <PacketCapturePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
   247→
   248→    /* Default fallback to placeholder */
   249→    return (
   250→      <LeafPlaceholder
   251→        moduleId={moduleId}

   254→      />

   257→
   258→  // state === "child" — children without grandchildren
   259→  switch (childId) {
   260→    case "services":
   261→      return <ServicesPage moduleId={moduleId} childId={childId} />;

   267→      return <DashboardConfPage moduleId={moduleId} childId={childId} />;
   268→    default:
   269→      return <SystemOverview moduleId={moduleId} />;
   270→  }

   311→function LevelBadge({ level }: { level: string }) {
   312→  if (level === "ERROR")
   313→    return <Badge className="bg-primary/10 text-primary hover:bg-primary/10">ERROR</Badge>;

   316→  return <Badge className="bg-emerald-500/10 text-emerald-700 hover:bg-emerald-500/10 dark:text-emerald-400">INFO</Badge>;
   317→}

   327→/** Standard 3-level breadcrumb builder for grandchild pages */
   328→function useBreadcrumb(moduleId: string, childId: string, grandchildId?: string) {
   329→  const { mod, child, grandchild } = useModuleHeader(moduleId, childId, grandchildId);
   330→  const setActive = useAppStore((s) => s.setActive);
   331→  return [
   332→    { label: "Cryptsk" },
   333→    { label: mod?.label ?? "System", onClick: () => setActive(moduleId, "", "") },
   334→    { label: child?.label ?? childId, onClick: () => setActive(moduleId, childId, "") },
   335→    ...(grandchild ? [{ label: grandchild.label }] : []),
   336→  ];
   337→}

   343→  onConfirm,

   345→  description,
   346→  busy,

   363→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={busy}>

   368→              <>

   370→              </>
   371→            ) : (
   372→              <>

   374→              </>
   375→            )}
   376→          </Button>
   377→        </DialogFooter>
   378→      </DialogContent>
   379→    </Dialog>

   970→    });
   971→  }, []);
