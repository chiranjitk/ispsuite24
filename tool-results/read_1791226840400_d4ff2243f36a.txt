    56→    56→export function PackageView({ moduleId, childId, grandchildId }: ViewProps) {
    57→    57→  const router = useViewRouter(moduleId, childId, grandchildId);
    58→    58→
    59→    59→  if (router.state === "loading") return null;
    60→    60→  if (router.state === "module-overview") return <PackageOverview moduleId={moduleId} />;
    61→    61→  if (router.state === "child-overview")
    62→    62→    return <ChildOverview moduleId={moduleId} childId={router.childId} />;
    63→    63→
    64→    64→  if (router.state === "grandchild") {
    65→    65→    // Bespoke grandchild handlers for the Package module
    66→    66→    if (childId === "package" && grandchildId === "create")
    67→    67→      return <CreatePackagePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    68→    68→    if (childId === "package" && grandchildId === "manage")
    69→    69→      return <ManagePackagesPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    70→    70→
    71→    71→    // Fall back to placeholder for other grandchildren
    72→    72→    return (
    73→    73→      <LeafPlaceholder
    74→    74→        moduleId={moduleId}
    75→    75→        childId={router.childId}
    76→    76→        grandchildId={router.grandchildId}
    77→    77→      />
    78→    78→    );
    79→    79→  }
    80→    80→
    81→    81→  // state === "child" — children without grandchildren
    82→    82→  switch (childId) {
    83→    83→    case "invoice":
    84→    84→      return <InvoiceChild moduleId={moduleId} childId={childId} />;
    85→    85→    case "invoice-template":
    86→    86→      return <InvoiceTemplateChild moduleId={moduleId} childId={childId} />;
    87→    87→    case "ancillary":
    88→    88→      return <AncillaryChild moduleId={moduleId} childId={childId} />;
    89→    89→    case "tax":
    90→    90→      return <TaxChild moduleId={moduleId} childId={childId} />;
    91→    91→    default:
    92→    92→      return <PackageOverview moduleId={moduleId} />;
    93→    93→  }
    94→    94→}
