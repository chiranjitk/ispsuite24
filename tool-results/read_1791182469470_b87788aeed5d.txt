    56→export function PackageView({ moduleId, childId, grandchildId }: ViewProps) {
    57→  const router = useViewRouter(moduleId, childId, grandchildId);
    58→
    59→  if (router.state === "loading") return null;
    60→  if (router.state === "module-overview") return <PackageOverview moduleId={moduleId} />;
    61→  if (router.state === "child-overview")
    62→    return <ChildOverview moduleId={moduleId} childId={router.childId} />;
    63→
    64→  if (router.state === "grandchild") {
    65→    // Bespoke grandchild handlers for the Package module
    66→    if (childId === "package" && grandchildId === "create")
    67→      return <CreatePackagePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    68→    if (childId === "package" && grandchildId === "manage")
    69→      return <ManagePackagesPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    70→
    71→    // Fall back to placeholder for other grandchildren
    72→    return (
    73→      <LeafPlaceholder
    74→        moduleId={moduleId}
    75→        childId={router.childId}
    76→        grandchildId={router.grandchildId}
    77→      />
    78→    );
    79→  }
    80→
    81→  // state === "child" — children without grandchildren
    82→  switch (childId) {
    83→    case "invoice":
    84→      return <InvoiceChild moduleId={moduleId} childId={childId} />;
    85→    case "invoice-template":
    86→      return <InvoiceTemplateChild moduleId={moduleId} childId={childId} />;
    87→    case "ancillary":
    88→      return <AncillaryChild moduleId={moduleId} childId={childId} />;
    89→    case "tax":
    90→      return <TaxChild moduleId={moduleId} childId={childId} />;
    91→    default:
    92→      return <PackageOverview moduleId={moduleId} />;
    93→  }
    94→}

   545→  const load = React.useCallback(() => {
   546→    setLoading(true);
   547→    packagesApi
   548→      .list({
   549→        search: search || undefined,
   550→        scheme: schemeFilter !== "all" ? schemeFilter : undefined,
   551→      })
   552→      .then((res) => {
   553→        setPackages(res.data ?? []);
   554→        setLoading(false);
   555→      });
   556→  }, [search, schemeFilter]);
   557→
   558→  React.useEffect(() => {
   559→    const t = setTimeout(load, 300);
   560→    return () => clearTimeout(t);
   561→  }, [load]);
   562→
   563→  const handleDelete = async () => {
   564→    if (!deleteTarget) return;
   565→    setDeleting(true);
   566→    const res = await packagesApi.delete(deleteTarget.groupid);
   567→    setDeleting(false);
   568→    if (res.responseCode === "0") {
   569→      toast({ title: "Package deleted", description: `"${deleteTarget.groupname}" was deleted.` });
   570→      setDeleteTarget(null);
   571→      load();
   572→    } else {
   573→      toast({
   574→        title: "Delete failed",
   575→        description: res.responseMsg,
   576→        variant: "destructive",
   577→      });
   578→    }
   579→  };
   580→
   581→  if (!mod || !child) return null;
   582→
   583→  return (
   584→    <div className="space-y-6">
   585→      <PageHeader
   586→        title={grandchild?.label ?? "Manage Packages"}
   587→        description="View, search and manage all billing plans. Data is backed by the packages API."
   588→        icon={<mod.icon className="h-5 w-5" />}
   589→        breadcrumb={[
   590→          { label: "Cryptsk" },
   591→          { label: mod.label, onClick: () => setActive(moduleId, "", "") },
   592→          { label: child.label, onClick: () => setActive(moduleId, childId, "") },
   593→          { label: grandchild?.label ?? "Manage" },
   594→        ]}
   595→        actions={
   596→          <Button onClick={() => setActive(moduleId, "package", "create")}>
   597→            <PlusCircle className="mr-2 h-4 w-4" /> Create Package
   598→          </Button>
   599→        }
   600→      />
   601→
   602→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   603→        <KpiCard label="Total Packages" value={String(packages.length)} icon={<PackageIcon className="h-4 w-4" />} accent />
   604→        <KpiCard label="Prepaid" value={String(packages.filter((p) => p.billingScheme === "PREPAID").length)} icon={<TrendingUp className="h-4 w-4" />} />
   605→        <KpiCard label="Postpaid" value={String(packages.filter((p) => p.billingScheme === "POSTPAID").length)} icon={<TrendingUp className="h-4 w-4" />} />
   606→        <KpiCard label="Active" value={String(packages.filter((p) => p.groupstatus === "Y").length)} icon={<PackageIcon className="h-4 w-4" />} />
   607→      </div>
   608→
   609→      <ActionBar>
   610→        <div className="relative flex-1 max-w-xs">
   611→          <Search className="pointer-events-none absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   612→          <Input
   613→            placeholder="Search packages…"
   614→            className="h-8 pl-8 text-xs"
   615→            value={search}
   616→            onChange={(e) => setSearch(e.target.value)}
   617→          />
   618→        </div>
   619→        <Select value={schemeFilter} onValueChange={setSchemeFilter}>
   620→          <SelectTrigger className="h-8 w-[140px]">
   621→            <SelectValue />
   622→          </SelectTrigger>
   623→          <SelectContent>
   624→            <SelectItem value="all">All schemes</SelectItem>
   625→            <SelectItem value="PREPAID">Prepaid</SelectItem>
   626→            <SelectItem value="POSTPAID">Postpaid</SelectItem>
   627→          </SelectContent>
   628→        </Select>
   629→        <Button variant="outline" size="sm" className="h-8" onClick={load}>
   630→          <Filter className="mr-1.5 h-3.5 w-3.5" /> Refresh
   631→        </Button>
   632→      </ActionBar>
   633→
   634→      <SectionCard>
   635→        {loading ? (
   636→          <div className="flex items-center justify-center py-12">
   637→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   638→          </div>
   639→        ) : packages.length === 0 ? (
   640→          <EmptyState
   641→            icon={<PackageIcon className="h-5 w-5" />}
   642→            title="No packages found"
   643→            description="Create your first package to get started."
   644→            action={
   645→              <Button onClick={() => setActive(moduleId, "package", "create")}>
   646→                <PlusCircle className="mr-2 h-4 w-4" /> Create Package
   647→              </Button>
   648→            }
   649→          />
   650→        ) : (
   651→          <div className="overflow-x-auto">
   652→            <Table>
   653→              <TableHeader>
   654→                <TableRow>
   655→                  <TableHead>ID</TableHead>
   656→                  <TableHead>Package Name</TableHead>
   657→                  <TableHead>Price</TableHead>
   658→                  <TableHead>Scheme</TableHead>
   659→                  <TableHead>Connection</TableHead>
   660→                  <TableHead>Cycle</TableHead>
   661→                  <TableHead>Duration</TableHead>
   662→                  <TableHead>Bind MAC</TableHead>
   663→                  <TableHead>Status</TableHead>
   664→                  <TableHead className="text-right">Actions</TableHead>
   665→                </TableRow>
   666→              </TableHeader>
   667→              <TableBody>
   668→                {packages.map((p) => (
   669→                  <TableRow key={p.groupid}>
   670→                    <TableCell className="font-mono text-xs text-muted-foreground">{p.groupid}</TableCell>
   671→                    <TableCell>
   672→                      <div>
   673→                        <p className="font-medium text-foreground">{p.groupname}</p>
   674→                        {p.description && (
   675→                          <p className="text-xs text-muted-foreground truncate max-w-[200px]">{p.description}</p>
   676→                        )}
   677→                      </div>
   678→                    </TableCell>
   679→                    <TableCell className="font-semibold">₹{p.price.toFixed(2)}</TableCell>
   680→                    <TableCell>
   681→                      <Badge
   682→                        variant={p.billingScheme === "PREPAID" ? "default" : "secondary"}
   683→                        className={p.billingScheme === "PREPAID" ? "bg-emerald-500/10 text-emerald-700 hover:bg-emerald-500/10 dark:text-emerald-400" : "bg-amber-500/10 text-amber-700 hover:bg-amber-500/10 dark:text-amber-400"}
   684→                      >
   685→                        {p.billingScheme}
   686→                      </Badge>
   687→                    </TableCell>
   688→                    <TableCell>
   689→                      <Badge variant="outline" className="text-[10px]">{p.connectionType}</Badge>
   690→                    </TableCell>
   691→                    <TableCell className="text-xs">{p.cycleType} ×{p.cyclemultiplier}</TableCell>
   692→                    <TableCell className="text-xs text-muted-foreground">{p.billDuration}h</TableCell>
   693→                    <TableCell>
   694→                      <Badge variant={p.isbindtomac === "Y" ? "default" : "secondary"} className="text-[10px]">
   695→                        {p.isbindtomac === "Y" ? "Yes" : "No"}
   696→                      </Badge>
   697→                    </TableCell>
   698→                    <TableCell>
   699→                      <Badge
   700→                        variant={p.groupstatus === "Y" ? "default" : "secondary"}
   701→                        className={p.groupstatus === "Y" ? "bg-emerald-500/10 text-emerald-700 hover:bg-emerald-500/10 dark:text-emerald-400" : ""}
   702→                      >
   703→                        {p.groupstatus === "Y" ? "Active" : "Inactive"}
   704→                      </Badge>
   705→                    </TableCell>
   706→                    <TableCell className="text-right">
   707→                      <div className="flex items-center justify-end gap-1">
   708→                        <Button
   709→                          variant="ghost"
   710→                          size="sm"
   711→                          onClick={() => toast({ title: "Edit package", description: `Editing ${p.groupname} (coming soon)` })}
   712→                        >
   713→                          <Pencil className="h-3.5 w-3.5" />
   714→                        </Button>
   715→                        <Button
   716→                          variant="ghost"
   717→                          size="sm"
   718→                          className="text-primary hover:text-primary"
   719→                          onClick={() => setDeleteTarget(p)}
   720→                        >
   721→                          <Trash2 className="h-3.5 w-3.5" />
   722→                        </Button>
   723→                      </div>
   724→                    </TableCell>
   725→                  </TableRow>
   726→                ))}
   727→              </TableBody>
   728→            </Table>
   729→          </div>
   730→        )}
   731→      </SectionCard>
   732→
   733→      {/* Delete confirmation dialog */}
   734→      <Dialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   735→        <DialogContent>
   736→          <DialogHeader>
   737→            <DialogTitle>Delete Package</DialogTitle>
   738→          </DialogHeader>
   739→          <p className="text-sm text-muted-foreground">
   740→            Are you sure you want to delete{" "}
   741→            <span className="font-medium text-foreground">{deleteTarget?.groupname}</span>?
   742→            This action cannot be undone.
   743→          </p>
   744→          <DialogFooter>
   745→            <Button variant="outline" onClick={() => setDeleteTarget(null)} disabled={deleting}>
   746→              Cancel
   747→            </Button>
   748→            <Button variant="destructive" onClick={handleDelete} disabled={deleting}>
   749→              {deleting ? (
   750→                <>
   751→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Deleting…
   752→                </>
   753→              ) : (
   754→                <>
   755→                  <Trash2 className="mr-2 h-4 w-4" /> Delete
   756→                </>
   757→              )}
   758→            </Button>
   759→          </DialogFooter>
   760→        </DialogContent>
   761→      </Dialog>
   762→    </div>
   763→  );
   764→}
