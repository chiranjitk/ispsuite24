   552→/** Special top-level views (not part of the 15 modules) */
   553→export interface TopLink {
   554→  id: string;
   555→  label: string;
   556→  icon: Icon;
   557→}
   558→
   559→export const TOP_LINKS: TopLink[] = [
   560→  { id: "home", label: "Home", icon: LayoutDashboard },
   561→  { id: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   562→];
   563→
   564→export function findModule(moduleId: string): NavModule | undefined {
   565→  return NAV_MODULES.find((m) => m.id === moduleId);
   566→}
   567→
   568→export function findChild(moduleId: string, childId: string): NavChild | undefined {
   569→  return findModule(moduleId)?.children.find((c) => c.id === childId);
   570→}
   571→
   572→export function findGrandchild(
   573→  moduleId: string,
   574→  childId: string,
   575→  grandchildId: string
   576→): NavGrandchild | undefined {
   577→  return findChild(moduleId, childId)?.grandchildren?.find(
   578→    (g) => g.id === grandchildId
   579→  );
   580→}
   581→
   582→/** Count all leaf pages (level-2 without grandchildren + all level-3 grandchildren) */
   583→export function countLeafPages(): number {
   584→  let count = 0;
   585→  for (const m of NAV_MODULES) {
   586→    for (const c of m.children) {
   587→      if (!c.grandchildren || c.grandchildren.length === 0) count += 1;
   588→      else count += c.grandchildren.length;
   589→    }
   590→  }
   591→  return count;
   592→}
