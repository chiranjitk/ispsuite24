    60→export function UserView({ moduleId, childId, grandchildId }: ViewProps) {
    61→  const router = useViewRouter(moduleId, childId, grandchildId);
    62→
    63→  if (router.state === "loading") return null;
    64→  if (router.state === "module-overview") return <UserOverview moduleId={moduleId} />;
    65→  if (router.state === "child-overview")
    66→    return <ChildOverview moduleId={moduleId} childId={router.childId} />;
    67→
    68→  if (router.state === "grandchild") {
    69→    // Bespoke grandchild handlers for the User module
    70→    if (childId === "manage-users" && grandchildId === "add-user")
    71→      return <AddUserPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    72→    if (childId === "manage-users" && grandchildId === "search-user")
    73→      return <SearchUserPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    74→    if (childId === "manage-users" && grandchildId === "advance-search")
    75→      return <SearchUserPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    76→    if (childId === "live-users" && grandchildId === "manage-live-users")
    77→      return <ManageLiveUsersPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    78→    if (childId === "live-users" && grandchildId === "search-live-users")
    79→      return <SearchLiveUsersPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    80→    if (childId === "zone" && grandchildId === "create-zone")
    81→      return <CreateZonePage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    82→    if (childId === "zone" && grandchildId === "manage-zone")
    83→      return <ManageZonesPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    84→    if (childId === "zone" && grandchildId === "search-zone-admin")
    85→      return <ManageZonesPage moduleId={moduleId} childId={childId} grandchildId={grandchildId} />;
    86→
    87→    return (
    88→      <LeafPlaceholder
    89→        moduleId={moduleId}
    90→        childId={router.childId}
    91→        grandchildId={router.grandchildId}
    92→      />
    93→    );
    94→  }
    95→
    96→  // state === "child" — children without grandchildren
    97→  switch (childId) {
    98→    case "zone":
    99→      return <ZoneManagement moduleId={moduleId} childId={childId} />;
   100→    case "pool":
   101→      return <PoolManagement moduleId={moduleId} childId={childId} />;
   102→    case "customers":
   103→      return <ManageCustomers moduleId={moduleId} childId={childId} />;
   104→    default:
   105→      return <UserOverview moduleId={moduleId} />;
   106→  }
   107→}
