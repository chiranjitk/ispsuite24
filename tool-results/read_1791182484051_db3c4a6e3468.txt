   250→function FirewallChild({ moduleId, childId }: ViewProps) {
   251→  const { mod, child } = useModuleHeader(moduleId, childId);
   252→  const { toast } = useToast();
   253→  const [rules, setRules] = React.useState<FirewallRule[]>(FIREWALL_RULES);
   254→  if (!mod) return null;
   255→  const toggle = (id: string) => {
   256→    setRules((prev) => prev.map((r) => (r.id === id ? { ...r, enabled: !r.enabled } : r)));
   257→    const r = rules.find((x) => x.id === id);
   258→    toast({
   259→      title: `${r?.enabled ? "Disabled" : "Enabled"} rule`,
   260→      description: r?.name,
   261→    });
   262→  };
   263→  const actionBadge = (action: FirewallRule["action"]) => {
   264→    if (action === "Allow")
   265→      return <Badge className="bg-emerald-500/10 text-emerald-700 hover:bg-emerald-500/10 dark:text-emerald-400">Allow</Badge>;
   266→    if (action === "Deny")
   267→      return <Badge className="bg-primary/10 text-primary hover:bg-primary/10">Deny</Badge>;
   268→    return <Badge className="bg-amber-500/10 text-amber-700 hover:bg-amber-500/10 dark:text-amber-400">Drop</Badge>;
   269→  };
   270→  return (
   271→    <div className="space-y-6">
   272→      <PageHeader
   273→        title={child!.label}
   274→        description="Create, manage and reorder firewall rules. DoS protection & free sites also configured here."
   275→        icon={<mod.icon className="h-5 w-5" />}
   276→        breadcrumb={[{ label: "Cryptsk" }, { label: mod.label }, { label: child!.label }]}
   277→        actions={
   278→          <Button onClick={() => toast({ title: "Create rule", description: "Firewall rule builder will open." })}>
   279→            <Plus className="mr-2 h-4 w-4" /> Create Rule
   280→          </Button>
   281→        }
   282→      />
   283→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   284→        <KpiCard label="Total Rules" value={String(rules.length)} icon={<ShieldAlert className="h-4 w-4" />} accent />
   285→        <KpiCard label="Enabled" value={String(rules.filter((r) => r.enabled).length)} icon={<Power className="h-4 w-4" />} />
   286→        <KpiCard label="Allow" value={String(rules.filter((r) => r.action === "Allow").length)} icon={<ArrowRight className="h-4 w-4" />} />
   287→        <KpiCard label="Deny/Drop" value={String(rules.filter((r) => r.action !== "Allow").length)} icon={<AlertTriangle className="h-4 w-4" />} />
   288→      </div>
   289→      <SectionCard title="Firewall Rules" description="Rules are evaluated top-to-bottom. Drag the position column to reorder.">
   290→        <div className="overflow-x-auto">
   291→          <Table>
   292→            <TableHeader>
   293→              <TableRow>
   294→                <TableHead className="w-16">Position</TableHead>
   295→                <TableHead>Name</TableHead>
   296→                <TableHead>Source</TableHead>
   297→                <TableHead>Destination</TableHead>
   298→                <TableHead>Service</TableHead>
   299→                <TableHead>Action</TableHead>
   300→                <TableHead>Enabled</TableHead>
   301→                <TableHead className="text-right">Actions</TableHead>
   302→              </TableRow>
   303→            </TableHeader>
   304→            <TableBody>
   305→              {rules.map((r) => (
   306→                <TableRow key={r.id} data-state={!r.enabled ? "selected" : undefined}>
   307→                  <TableCell>
   308→                    <div className="flex items-center gap-1 font-mono text-xs text-muted-foreground">
   309→                      <GripIcon /> {String(r.position).padStart(2, "0")}
   310→                    </div>
   311→                  </TableCell>
   312→                  <TableCell className="font-medium">{r.name}</TableCell>
   313→                  <TableCell className="font-mono text-xs text-muted-foreground">{r.src}</TableCell>
   314→                  <TableCell className="font-mono text-xs text-muted-foreground">{r.dst}</TableCell>
   315→                  <TableCell className="font-mono text-xs text-muted-foreground">{r.service}</TableCell>
   316→                  <TableCell>{actionBadge(r.action)}</TableCell>
   317→                  <TableCell>
   318→                    <Checkbox checked={r.enabled} onCheckedChange={() => toggle(r.id)} aria-label={`Toggle ${r.name}`} />
   319→                  </TableCell>
   320→                  <TableCell className="text-right">
   321→                    <Button variant="ghost" size="sm" onClick={() => toast({ title: "Edit rule", description: r.name })}>
   322→                      <Pencil className="mr-1 h-3.5 w-3.5" /> Edit
   323→                    </Button>
   324→                  </TableCell>
   325→                </TableRow>
   326→              ))}
   327→            </TableBody>
   328→          </Table>
   329→        </div>
   330→      </SectionCard>
   331→    </div>
   332→  );

   472→function ServicesChild({ moduleId, childId }: ViewProps) {
   473→  const { mod, child } = useModuleHeader(moduleId, childId);
   474→  const { toast } = useToast();
   475→  const [services, setServices] = React.useState<SystemService[]>(SYSTEM_SERVICES);
   476→  if (!mod) return null;
   477→  const toggleService = (id: string) => {
   478→    setServices((prev) => prev.map((s) => (s.id === id ? { ...s, running: !s.running } : s)));
   479→    const s = services.find((x) => x.id === id);
   480→    toast({
   481→      title: `${s?.running ? "Stopped" : "Started"} service`,
   482→      description: s?.name,
   483→      variant: s?.running ? "destructive" : "default",
   484→    });
   485→  };
   486→  const act = (id: string, action: "start" | "stop" | "restart") => {
   487→    const s = services.find((x) => x.id === id);
   488→    toast({
   489→      title: `${action.charAt(0).toUpperCase() + action.slice(1)} ${s?.name}`,
