    55→export function PaymentTrackingView({ moduleId, childId }: ViewProps) {
    56→  const { mod } = useModuleHeader(moduleId, childId);
    57→  if (!mod) return null;
    58→  if (!childId) return <PaymentTrackingOverview moduleId={moduleId} />;
    59→  switch (childId) {
    60→    case "manage-accounts":
    61→      return <ManageAccounts moduleId={moduleId} childId={childId} />;
    62→    case "search-accounts":
    63→      return <SearchAccounts moduleId={moduleId} childId={childId} />;
    64→    case "payment-details":
    65→      return <PaymentDetails moduleId={moduleId} childId={childId} />;
    66→    case "reverse":
    67→      return <ReverseTransactions moduleId={moduleId} childId={childId} />;
    68→    case "settle":
    69→      return <SettleTransactions moduleId={moduleId} childId={childId} />;
    70→    default:
    71→      return <PaymentTrackingOverview moduleId={moduleId} />;
    72→  }
    73→}

   135→function ManageAccounts({ moduleId, childId }: ViewProps) {
   136→  const { toast } = useToast();
   137→  const [modes, setModes] = React.useState(PAYMENT_MODES);
   138→  const toggleMode = (id: string, enabled: boolean) => {
   139→    setModes((prev) => prev.map((m) => (m.id === id ? { ...m, enabled } : m)));
   140→    const m = modes.find((x) => x.id === id);
   141→    toast({ title: `${m?.name} ${enabled ? "enabled" : "disabled"}` });
   142→  };
   143→  const setDefault = (id: string) => {
   144→    setModes((prev) => prev.map((m) => ({ ...m, isDefault: m.id === id })));
   145→    const m = modes.find((x) => x.id === id);

   464→/* ---------------- Reverse Transactions ---------------- */
   465→
   466→function ReverseTransactions({ moduleId, childId }: ViewProps) {
   467→  const { toast } = useToast();
   468→  const [selected, setSelected] = React.useState<Set<string>>(new Set());
   469→  const [reason, setReason] = React.useState("");
   470→
   471→  // show all, but pre-flag reversed ones; allow reversing pending/settled ones
   472→  const rows = PAYMENT_TXNS;
   473→  const toggle = (id: string) => {
   474→    const next = new Set(selected);
   475→    if (next.has(id)) next.delete(id); else next.add(id);
   476→    setSelected(next);

   483→    }
   484→    if (!reason.trim()) {
   485→      toast({ title: "Reason required", description: "Provide a reason for reversal.", variant: "destructive" });
   486→      return;
   487→    }
   488→    toast({ title: "Transactions reversed", description: `${selected.size} transaction(s) reversed.`, variant: "destructive" });

   490→    setReason("");

   512→                <TableHead>Date</TableHead>
   513→                <TableHead>Status</TableHead>
   514→                <TableHead>Collected By</TableHead>
   515→              </TableRow>
   516→            </TableHeader>
   517→            <TableBody>
   518→              {rows.map((t) => (
   519→                <TableRow key={t.id} data-state={selected.has(t.id) ? "selected" : undefined}>
   520→                  <TableCell>
   521→                    <Checkbox checked={selected.has(t.id)} onCheckedChange={() => toggle(t.id)} />

   534→                  <TableCell className="text-xs text-muted-foreground">{t.collectedBy}</TableCell>

   536→              ))}

   544→          </div>
   545→          <Button variant="destructive" onClick={reverse}>
   546→            <Undo2 className="mr-1.5 h-4 w-4" /> Reverse Selected
   547→          </Button>
   548→        </div>
   549→      </SectionCard>
   550→    </div>
   551→  );

   554→/* ---------------- Settle Transactions ---------------- */
   555→
   556→function SettleTransactions({ moduleId, childId }: ViewProps) {
   557→  const { toast } = useToast();
   558→  const pending = PAYMENT_TXNS.filter((t) => t.status === "Pending");
   559→  const [selected, setSelected] = React.useState<Set<string>>(new Set());
   560→  const toggle = (id: string) => {
   561→    const next = new Set(selected);
   562→    if (next.has(id)) next.delete(id); else next.add(id);
   563→    setSelected(next);
   564→  };
   565→  const toggleAll = () => {
   566→    if (selected.size === pending.length) setSelected(new Set());
   567→    else setSelected(new Set(pending.map((p) => p.id)));
   568→  };
   569→  const settle = () => {
   570→    if (selected.size === 0) {
   571→      toast({ title: "No selection", description: "Select one or more transactions to settle.", variant: "destructive" });
   572→      return;
   573→    }
   574→    toast({ title: "Transactions settled", description: `${selected.size} transaction(s) settled.` });
   575→    setSelected(new Set());
   576→  };
   577→  return (
   578→    <div className="space-y-6">
   579→      <ChildHeader moduleId={moduleId} childId={childId} />
   580→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   581→        <KpiCard label="Pending Settlement" value={String(pending.length)} icon={<Clock className="h-4 w-4" />} accent />
   582→        <KpiCard label="Selected" value={String(selected.size)} icon={<CheckCircle2 className="h-4 w-4" />} />
   583→        <KpiCard label="Pending Amount" value={`₹ ${pending.reduce((s, t) => s + parseFloat(t.amount), 0).toFixed(2)}`} icon={<Coins className="h-4 w-4" />} />
   584→      </div>
   585→      <SectionCard title="Pending Transactions" description="Select transactions to settle">
   586→        {pending.length === 0 ? (
   587→          <EmptyState icon={<CheckCircle2 className="h-5 w-5" />} title="No pending transactions" description="Everything is settled." />
   588→        ) : (
   589→          <>
   590→            <ActionBar>
   591→              <Button size="sm" onClick={settle}>
   592→                <CheckCircle2 className="mr-1.5 h-3.5 w-3.5" /> Settle Selected
   593→              </Button>
   594→              <Button variant="outline" size="sm" onClick={toggleAll}>
   595→                {selected.size === pending.length ? "Clear All" : "Select All"}
   596→              </Button>
   597→            </ActionBar>
   598→            <div className="mt-3 overflow-x-auto scrollbar-thin">
   599→              <Table>
   600→                <TableHeader>
   601→                  <TableRow>
   602→                    <TableHead className="w-8">
   603→                      <Checkbox
   604→                        checked={pending.length > 0 && selected.size === pending.length}
   605→                        onCheckedChange={toggleAll}
   606→                      />
   607→                    </TableHead>
   608→                    <TableHead>Txn ID</TableHead>
   609→                    <TableHead>Account</TableHead>
   610→                    <TableHead>Customer</TableHead>
   611→                    <TableHead>Mode</TableHead>
   612→                    <TableHead>Amount</TableHead>
   613→                    <TableHead>Date</TableHead>
   614→                    <TableHead>Collected By</TableHead>
   615→                  </TableRow>
   616→                </TableHeader>
   617→                <TableBody>
   618→                  {pending.map((t) => (
   619→                    <TableRow key={t.id} data-state={selected.has(t.id) ? "selected" : undefined}>
   620→                      <TableCell>
   621→                        <Checkbox checked={selected.has(t.id)} onCheckedChange={() => toggle(t.id)} />
   622→                      </TableCell>
   623→                      <TableCell className="font-mono text-xs">{t.id}</TableCell>
   624→                      <TableCell className="font-mono text-xs">{t.account}</TableCell>
   625→                      <TableCell className="font-medium text-primary">{t.customer}</TableCell>
   626→                      <TableCell>
   627→                        <Badge variant="outline" className={`text-[10px] ${modeBadgeClass(t.mode)}`}>{t.mode}</Badge>
   628→                      </TableCell>
   629→                      <TableCell className="font-mono text-xs">{t.amount}</TableCell>
   630→                      <TableCell className="text-xs text-muted-foreground">{t.date}</TableCell>
   631→                      <TableCell className="text-xs text-muted-foreground">{t.collectedBy}</TableCell>
   632→                    </TableRow>
   633→                  ))}
   634→                </TableBody>
   635→              </Table>
   636→            </div>
   637→          </>
   638→        )}
   639→      </SectionCard>
   640→    </div>
   641→  );
