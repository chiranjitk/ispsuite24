    55→export function HelpView({ moduleId, childId }: ViewProps) {
    56→  const { mod } = useModuleHeader(moduleId, childId);
    57→  if (!mod) return null;
    58→  if (!childId) return <HelpOverview moduleId={moduleId} />;
    59→  switch (childId) {
    60→    case "company-info":
    61→      return <CompanyInfo moduleId={moduleId} childId={childId} />;
    62→    case "client":
    63→      return <ClientApp moduleId={moduleId} childId={childId} />;
    64→    case "upgrade":
    65→      return <UpgradeVersion moduleId={moduleId} childId={childId} />;
    66→    case "register":
    67→      return <RegisterProduct moduleId={moduleId} childId={childId} />;
    68→    case "customization":
    69→      return <ManageCustomization moduleId={moduleId} childId={childId} />;
    70→    case "documentation":
    71→      return <Documentation moduleId={moduleId} childId={childId} />;
    72→    case "about":
    73→      return <About moduleId={moduleId} childId={childId} />;
    74→    default:
    75→      return <HelpOverview moduleId={moduleId} />;
    76→  }
    77→}

   445→/* ---------------- Documentation ---------------- */
   446→
   447→function DocBlockRenderer({ block }: { block: DocBlock }) {
   448→  switch (block.type) {
   449→    case "h1":
   450→      return <h1 className="text-xl font-semibold tracking-tight text-foreground">{block.text}</h1>;
   451→    case "h2":
   452→      return <h2 className="mt-4 text-base font-semibold text-foreground">{block.text}</h2>;
   453→    case "p":
   454→      return <p className="text-sm leading-relaxed text-muted-foreground">{block.text}</p>;
   455→    case "ul":
   456→      return (
   457→        <ul className="ml-5 list-disc space-y-1 text-sm text-muted-foreground">
   458→          {block.items?.map((i, idx) => <li key={idx}>{i}</li>)}
   459→        </ul>
   460→      );
   461→    case "code":
   462→      return (
   463→        <pre className="overflow-x-auto rounded-md border border-border bg-muted/60 p-3 font-mono text-xs text-foreground scrollbar-thin">
   464→          <code>{block.text}</code>
   465→        </pre>
   466→      );
   467→    default:
   468→      return null;
   469→  }
   470→}
   471→
   472→function Documentation({ moduleId, childId }: ViewProps) {
   473→  const { toast } = useToast();
   474→  const [activeId, setActiveId] = React.useState(DOC_SECTIONS[0].id);
   475→  const [query, setQuery] = React.useState("");
   476→  const filtered = DOC_SECTIONS.filter((s) =>
   477→    !query.trim()
   478→      ? true
   479→      : s.title.toLowerCase().includes(query.toLowerCase()) ||
   480→        s.content.some((b) => (b.text ?? "").toLowerCase().includes(query.toLowerCase()))
   481→  );
   482→  const active = DOC_SECTIONS.find((s) => s.id === activeId) ?? DOC_SECTIONS[0];
   483→
   484→  return (
   485→    <div className="space-y-6">
   486→      <ChildHeader moduleId={moduleId} childId={childId} />
   487→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-5">
   488→        <SectionCard title="Sections" description={`${DOC_SECTIONS.length} articles`} className="lg:col-span-1">
   489→          <div className="relative mb-3">
   490→            <Search className="pointer-events-none absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   491→            <Input placeholder="Search docs…" className="h-8 pl-8 text-xs" value={query} onChange={(e) => setQuery(e.target.value)} />
   492→          </div>
   493→          <div className="flex flex-col gap-1">
   494→            {filtered.length === 0 ? (
   495→              <EmptyState icon={<Search className="h-4 w-4" />} title="No results" />
   496→            ) : (
   497→              filtered.map((s) => (
   498→                <button
   499→                  key={s.id}
   500→                  onClick={() => setActiveId(s.id)}
   501→                  className={
   502→                    "flex items-center gap-2 rounded-md px-3 py-2 text-left text-sm transition-colors " +
   503→                    (activeId === s.id
   504→                      ? "bg-primary/10 font-medium text-primary"
   505→                      : "text-muted-foreground hover:bg-muted hover:text-foreground")
   506→                  }
   507→                >
   508→                  <FileText className="h-3.5 w-3.5 shrink-0" />
   509→                  <span className="truncate">{s.title}</span>
   510→                </button>
   511→              ))
   512→            )}
   513→          </div>
   514→        </SectionCard>
   515→        <SectionCard
   516→          title={active.title}
   517→          description={active.category}
   518→          className="lg:col-span-4"
   519→          actions={
   520→            <Button variant="outline" size="sm" onClick={() => toast({ title: "Exported", description: `${active.title}.pdf download started.` })}>
   521→              <DownloadCloud className="mr-1.5 h-3.5 w-3.5" /> Export PDF
   522→            </Button>
   523→          }
   524→        >
   525→          <div className="space-y-3">
   526→            {active.content.map((b, i) => <DocBlockRenderer key={i} block={b} />)}
   527→          </div>
   528→        </SectionCard>
   529→      </div>
   530→    </div>
   531→  );
   532→}
