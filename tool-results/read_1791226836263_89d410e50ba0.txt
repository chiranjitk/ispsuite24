    59→export function PolicyView({ moduleId, childId, grandchildId }: ViewProps) {
    60→  const router = useViewRouter(moduleId, childId, grandchildId);
    61→
    62→  if (router.state === "loading") return null;
    63→  if (router.state === "module-overview")
    64→    return <PolicyOverview moduleId={moduleId} />;
    65→  if (router.state === "child-overview")
    66→    return <ChildOverview moduleId={moduleId} childId={router.childId} />;
    67→
    68→  if (router.state === "grandchild") {
    69→    // Surfing Quota
    70→    if (childId === "surfing-quota" && grandchildId === "create")
    71→      return <SurfCreatePage {...{ moduleId, childId, grandchildId }} />;
    72→    if (childId === "surfing-quota" && grandchildId === "manage")
    73→      return <SurfManagePage {...{ moduleId, childId, grandchildId }} />;
    74→
    75→    // Access Time
    76→    if (childId === "access-time" && grandchildId === "create")
    77→      return <AccessCreatePage {...{ moduleId, childId, grandchildId }} />;
    78→    if (childId === "access-time" && grandchildId === "manage")
    79→      return <AccessManagePage {...{ moduleId, childId, grandchildId }} />;
    80→
    81→    // Bandwidth
    82→    if (childId === "bandwidth" && grandchildId === "create")
    83→      return <BwCreatePage {...{ moduleId, childId, grandchildId }} />;
    84→    if (childId === "bandwidth" && grandchildId === "manage")
    85→      return <BwManagePage {...{ moduleId, childId, grandchildId }} />;
    86→    if (childId === "bandwidth" && grandchildId === "schedule")
    87→      return <BwSchedulePage {...{ moduleId, childId, grandchildId }} />;
    88→
    89→    // Data Transfer
    90→    if (childId === "data-transfer" && grandchildId === "create")
    91→      return <DtCreatePage {...{ moduleId, childId, grandchildId }} />;
    92→    if (childId === "data-transfer" && grandchildId === "manage")
    93→      return <DtManagePage {...{ moduleId, childId, grandchildId }} />;
    94→
    95→    // FAP
    96→    if (childId === "fap" && grandchildId === "create-fap")
    97→      return <FapCreatePage {...{ moduleId, childId, grandchildId }} />;
    98→    if (childId === "fap" && grandchildId === "manage-fap")
    99→      return <FapManagePage {...{ moduleId, childId, grandchildId }} />;
   100→
   101→    // QoS
   102→    if (childId === "qos" && grandchildId === "create-cache")
   103→      return <QosCreatePage {...{ moduleId, childId, grandchildId }} />;
   104→    if (childId === "qos" && grandchildId === "manage-cache")
   105→      return <QosManagePage {...{ moduleId, childId, grandchildId }} />;
   106→
   107→    return (
   108→      <LeafPlaceholder
   109→        moduleId={moduleId}
   110→        childId={router.childId}
   111→        grandchildId={router.grandchildId}
   112→      />
   113→    );
   114→  }
   115→
   116→  return <PolicyOverview moduleId={moduleId} />;
   117→}

   207→function SurfCreatePage({ moduleId, childId, grandchildId }: ViewProps) {

   209→  const { toast } = useToast();

   211→  const [saving, setSaving] = React.useState(false);
   212→  const [form, setForm] = React.useState({
   213→    policyName: "",
   214→    policyType: "Time-based",
   215→    allotedhrs: "1",
   216→    allotedmin: "00",
   217→    unlimitedtime: false,
   218→    sessionpulse: "0",
   219→    setexpireallotedhrs: "0",
   220→    setexpireallotedmin: "00",
   221→    setexpirealloteddays: "30",
   222→    unlimitedvalue: false,
   223→    cycletype: "Daily",
   224→    cycleallotedhrs: "0",
   225→    cycleallotedmin: "00",
   226→    description: "",
   227→  });

   231→    e.preventDefault();

   236→      policyType: form.policyType,
   237→      timeAllowed: form.unlimitedtime ? "Unlimited" : `${form.allotedhrs}:${form.allotedmin}`,
   238→      expirationDuration: form.unlimitedvalue ? "Unlimited" : form.setexpirealloteddays,
   239→      sessionPulse: form.sessionpulse,
   240→      description: form.description,
   241→      status: "Active",
   242→    });

   244→    toast({ title: "Surfing policy created", description: form.policyName });

   248→  return (
   249→    <div className="space-y-6">
   250→      <PageHeader title="Create Surfing Quota Policy" description="Create a new surfing quota policy (PolicyManager servlet)." icon={<ShieldCheck className="h-5 w-5" />} breadcrumb={breadcrumb}

   255→            <div className="space-y-2">
   256→              <Label>Policy Name <span className="text-primary">*</span></Label>
   257→              <Input value={form.policyName} onChange={(e) => set("policyName", e.target.value)} placeholder="e.g. 1 Hour Daily" required />
   258→            </div>
   259→            <div className="space-y-2">
   260→              <Label>Policy Type <span className="text-primary">*</span></Label>
   261→              <Select value={form.policyType} onValueChange={(v) => set("policyType", v)}>
   262→                <SelectTrigger><SelectValue /></SelectTrigger>
   263→                <SelectContent><SelectItem value="Time-based">Time-based</SelectItem><SelectItem value="Unlimited">Unlimited</SelectItem></SelectContent>
   264→              </Select>

   327→function SurfManagePage({ moduleId, childId, grandchildId }: ViewProps) {
   328→  const breadcrumb = useBreadcrumb(moduleId, childId, grandchildId);
   329→  const { toast } = useToast();
   330→  const setActive = useAppStore((s) => s.setActive);
   331→  const [rows, setRows] = React.useState<any[]>([]);
   332→  const [loading, setLoading] = React.useState(true);
   333→  const [selected, setSelected] = React.useState<Set<number>>(new Set());
   334→
   335→  const load = React.useCallback(() => {
   336→    setLoading(true);
   337→    policyApi.list("surfingPolicies").then((res) => { setRows(res.data ?? []); setLoading(false); });
   338→  }, []);
   339→  React.useEffect(() => { load(); }, [load]);
   340→
   341→  const handleDelete = async () => {
   342→    for (const id of selected) await policyApi.delete("surfingPolicies", id);
   343→    toast({ title: "Policies deleted", description: `${selected.size} policy(s) deleted.` });
   344→    setSelected(new Set());
   345→    load();
   346→  };
   347→
   348→  return (
   349→    <div className="space-y-6">
   350→      <PageHeader title="Manage Surfing Quota Policy" description="View, and delete surfing quota policies." icon={<ShieldCheck className="h-5 w-5" />} breadcrumb={breadcrumb}
   351→        actions={<Button onClick={() => setActive(moduleId, "surfing-quota", "create")}><Plus className="mr-2 h-4 w-4" /> Create</Button>} />
   352→      {loading ? <PageLoader /> : (
   353→        <SectionCard title="Surfing Quota Policies" description={`${rows.length} policies`}>
   354→          <div className="overflow-x-auto">
   355→            <Table>
   356→              <TableHeader><TableRow>
   357→                <TableHead>Policy Name</TableHead><TableHead>Policy Type</TableHead><TableHead>Time Allowed (HH:mm)</TableHead>
   358→                <TableHead>Expiration Duration (days)</TableHead><TableHead>Session Pulse (minutes)</TableHead><TableHead>Description</TableHead><TableHead>Del</TableHead>
   359→              </TableRow></TableHeader>
   360→              <TableBody>
   361→                {rows.map((r) => (
   362→                  <TableRow key={r.id}>
   363→                    <TableCell className="font-medium">{r.policyName}</TableCell>
   364→                    <TableCell><Badge variant="outline" className="text-[10px]">{r.policyType}</Badge></TableCell>
   365→                    <TableCell className="font-mono text-xs">{r.timeAllowed}</TableCell>
   366→                    <TableCell>{r.expirationDuration}</TableCell>
   367→                    <TableCell>{r.sessionPulse}</TableCell>
   368→                    <TableCell className="text-xs text-muted-foreground">{r.description || "—"}</TableCell>
   369→                    <TableCell><Checkbox checked={selected.has(r.id)} onCheckedChange={() => { const n = new Set(selected); if (n.has(r.id)) n.delete(r.id); else n.add(r.id); setSelected(n); }} /></TableCell>
   370→                  </TableRow>
   371→                ))}
   372→              </TableBody>
   373→            </Table>
   374→          </div>
   375→          {selected.size > 0 && <div className="mt-3 flex justify-end"><Button variant="destructive" size="sm" onClick={handleDelete}><Trash2 className="mr-2 h-3.5 w-3.5" /> Delete Policy ({selected.size})</Button></div>}
   376→        </SectionCard>
   377→      )}
   378→    </div>
   379→  );
   380→}
