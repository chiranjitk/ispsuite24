    41→function rand<T>(arr: T[]): T {
    42→  return arr[Math.floor(Math.random() * arr.length)];
    43→}
    44→function pad(n: number, len: number) {
    45→  return String(n).padStart(len, "0");
    46→}
    47→function mac() {
    48→  const hex = "0123456789ABCDEF";
    49→  let s = "";
    50→  for (let i = 0; i < 6; i++) {
    51→    s += hex[Math.floor(Math.random() * 16)] + hex[Math.floor(Math.random() * 16)];
    52→    if (i < 5) s += ":";
    53→  }
    54→  return s;
    55→}
