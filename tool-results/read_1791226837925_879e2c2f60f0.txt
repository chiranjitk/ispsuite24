    59→    59→export function PolicyView({ moduleId, childId, grandchildId }: ViewProps) {
    60→    60→  const router = useViewRouter(moduleId, childId, grandchildId);
    61→    61→
    62→    62→  if (router.state === "loading") return null;
    63→    63→  if (router.state === "module-overview")
    64→    64→    return <PolicyOverview moduleId={moduleId} />;
    65→    65→  if (router.state === "child-overview")
    66→    66→    return <ChildOverview moduleId={moduleId} childId={router.childId} />;
    67→    67→
    68→    68→  if (router.state === "grandchild") {
    69→    69→    // Surfing Quota
    70→    70→    if (childId === "surfing-quota" && grandchildId === "create")
    71→    71→      return <SurfCreatePage {...{ moduleId, childId, grandchildId }} />;
    72→    72→    if (childId === "surfing-quota" && grandchildId === "manage")
    73→    73→      return <SurfManagePage {...{ moduleId, childId, grandchildId }} />;
    74→    74→
    75→    75→    // Access Time
    76→    76→    if (childId === "access-time" && grandchildId === "create")
    77→    77→      return <AccessCreatePage {...{ moduleId, childId, grandchildId }} />;
    78→    78→    if (childId === "access-time" && grandchildId === "manage")
    79→    79→      return <AccessManagePage {...{ moduleId, childId, grandchildId }} />;
    80→    80→
    81→    81→    // Bandwidth
    82→    82→    if (childId === "bandwidth" && grandchildId === "create")
    83→    83→      return <BwCreatePage {...{ moduleId, childId, grandchildId }} />;
    84→    84→    if (childId === "bandwidth" && grandchildId === "manage")
    85→    85→      return <BwManagePage {...{ moduleId, childId, grandchildId }} />;
    86→    86→    if (childId === "bandwidth" && grandchildId === "schedule")
    87→    87→      return <BwSchedulePage {...{ moduleId, childId, grandchildId }} />;
    88→    88→
    89→    89→    // Data Transfer
    90→    90→    if (childId === "data-transfer" && grandchildId === "create")
    91→    91→      return <DtCreatePage {...{ moduleId, childId, grandchildId }} />;
    92→    92→    if (childId === "data-transfer" && grandchildId === "manage")
    93→    93→      return <DtManagePage {...{ moduleId, childId, grandchildId }} />;
    94→    94→
    95→    95→    // FAP
    96→    96→    if (childId === "fap" && grandchildId === "create-fap")
    97→    97→      return <FapCreatePage {...{ moduleId, childId, grandchildId }} />;
    98→    98→    if (childId === "fap" && grandchildId === "manage-fap")
    99→    99→      return <FapManagePage {...{ moduleId, childId, grandchildId }} />;
   100→   100→
   101→   101→    // QoS
   102→   102→    if (childId === "qos" && grandchildId === "create-cache")
   103→   103→      return <QosCreatePage {...{ moduleId, childId, grandchildId }} />;
   104→   104→    if (childId === "qos" && grandchildId === "manage-cache")
   105→   105→      return <QosManagePage {...{ moduleId, childId, grandchildId }} />;
   106→   106→
   107→   107→    return (
   108→   108→      <LeafPlaceholder
   109→   109→        moduleId={moduleId}
   110→   110→        childId={router.childId}
   111→   111→        grandchildId={router.grandchildId}

   327→   327→function SurfManagePage({ moduleId, childId, grandchildId }: ViewProps) {
   328→   328→  const breadcrumb = useBreadcrumb(moduleId, childId, grandchildId);
   329→   329→  const { toast } = useToast();
   330→   330→  const setActive = useAppStore((s) => s.setActive);
   331→   331→  const [rows, setRows] = React.useState<any[]>([]);
   332→   332→  const [loading, setLoading] = React.useState(true);
   333→   333→  const [selected, setSelected] = React.useState<Set<number>>(new Set());
   334→   334→
   335→   335→  const load = React.useCallback(() => {
   336→   336→    setLoading(true);
   337→   337→    policyApi.list("surfingPolicies").then((res) => { setRows(res.data ?? []); setLoading(false); });
   338→   338→  }, []);
   339→   339→  React.useEffect(() => { load(); }, [load]);
   340→   340→
   341→   341→  const handleDelete = async () => {
   342→   342→    for (const id of selected) await policyApi.delete("surfingPolicies", id);
   343→   343→    toast({ title: "Policies deleted", description: `${selected.size} policy(s) deleted.` });
   344→   344→    setSelected(new Set());
   345→   345→    load();
   346→   346→  };
   347→   347→
   348→   348→  return (
   349→   349→    <div className="space-y-6">
   350→   350→      <PageHeader title="Manage Surfing Quota Policy" description="View, and delete surfing quota policies." icon={<ShieldCheck className="h-5 w-5" />} breadcrumb={breadcrumb}
   351→   351→        actions={<Button onClick={() => setActive(moduleId, "surfing-quota", "create")}><Plus className="mr-2 h-4 w-4" /> Create</Button>} />
   352→   352→      {loading ? <PageLoader /> : (
   353→   353→        <SectionCard title="Surfing Quota Policies" description={`${rows.length} policies`}>
   354→   354→          <div className="overflow-x-auto">
   355→   355→            <Table>
   356→   356→              <TableHeader><TableRow>
   357→   357→                <TableHead>Policy Name</TableHead><TableHead>Policy Type</TableHead><TableHead>Time Allowed (HH:mm)</TableHead>
   358→   358→                <TableHead>Expiration Duration (days)</TableHead><TableHead>Session Pulse (minutes)</TableHead><TableHead>Description</TableHead><TableHead>Del</TableHead>
   359→   359→              </TableRow></TableHeader>
   360→   360→              <TableBody>
   361→   361→                {rows.map((r) => (
   362→   362→                  <TableRow key={r.id}>
   363→   363→                    <TableCell className="font-medium">{r.policyName}</TableCell>
   364→   364→                    <TableCell><Badge variant="outline" className="text-[10px]">{r.policyType}</Badge></TableCell>
   365→   365→                    <TableCell className="font-mono text-xs">{r.timeAllowed}</TableCell>
   366→   366→                    <TableCell>{r.expirationDuration}</TableCell>
   367→   367→                    <TableCell>{r.sessionPulse}</TableCell>
   368→   368→                    <TableCell className="text-xs text-muted-foreground">{r.description || "—"}</TableCell>
   369→   369→                    <TableCell><Checkbox checked={selected.has(r.id)} onCheckedChange={() => { const n = new Set(selected); if (n.has(r.id)) n.delete(r.id); else n.add(r.id); setSelected(n); }} /></TableCell>
   370→   370→                  </TableRow>
   371→   371→                ))}
   372→   372→              </TableBody>
   373→   373→            </Table>
   374→   374→          </div>
   375→   375→          {selected.size > 0 && <div className="mt-3 flex justify-end"><Button variant="destructive" size="sm" onClick={handleDelete}><Trash2 className="mr-2 h-3.5 w-3.5" /> Delete Policy ({selected.size})</Button></div>}
   376→   376→        </SectionCard>
   377→   377→      )}
   378→   378→    </div>
   379→   379→  );
   380→   380→}
